\begin{dang}
\textbf{Exercise 1}. Let $U=k^2-\{0\}$, show that the set of regular functions $U\to k$ are polynomials, then 
conclude that $U$ is not an affine variety.    
\end{dang}
\textbf{Solution:} Let $\varphi\in\mathcal{O}_U(U)$ be a regular function, then 
$\varphi|_{D_{k^2}(X)}=f/X^m$ and $\varphi|_{D_{k^2}(Y)}=g/Y^n$ and on the intersection $D_{k^2}(X)\cap D_{k^2}(Y)=D_{k^2}(XY)$, we have 
\begin{equation}
    Y^nf - X^mg = 0\quad(*).
\end{equation}
Since there exist an infinity number of points $p\in D_{k^2}(XY)$ (assuming $k$ is algebraically closed), 
we also have the equality $(*)$ in $k[X,Y]$. Now, let $f=\sum a_{ij}X^iY^j$ and 
$g=\sum b_{ij} X^iY^j$. Then, equality $(*)$ means 
\begin{equation}
    a_{i, j-n} = a_{i-m,j},\;\forall i\geq m, j\geq n.
\end{equation}  
Therefore $f/X^m = g/Y^n\in k[X,Y]$ and $\varphi$ is a polynomial, because $U=D_{k^2}(X)\cup D_{k^2}(Y)$. In conclusion,
we proved that the inclusion $i:U\hookrightarrow k^2$ induces a ring isomorphism
\begin{equation}
    i^\#(k^2): k[X,Y]\to \mathcal{O}_U(U):f\mapsto f|_U\; (**).
\end{equation}
Now assume $U=V(S),\;S\subset k[X,Y]$, then $f\in S\Rightarrow f|_U=0$ and $f=0$ because of $(**)$. So 
$S=\{0\}$ and $U=k^2$, which is a contradiction.\qed


\begin{dang}
   \textbf{Exercise 2.} Let \(V = V(XW - YZ) \subset \mathbb{A}^4_k\), such that \(k[V] = k[X, Y, Z, W]/(XW - YZ)\).
Let \(\overline{X}, \overline{Y}, \overline{Z}, \overline{W}\) be the residue classes of \(X, Y, Z, W\) in \(k[V]\). 
Then \(\overline{X}/\overline{Y} = \overline{Z}/\overline{W} = f \in k(V)\)
is defined at \(P = (x, y, z, w) \in V\) if \(y \neq 0\) or \(w \neq 0\). Show that it is impossible to write
\(f = F/G\), with \(F, G \in k[V]\) and \(G(p) \neq 0\) for every point \(p\) where \(G\) is defined. Show that the
set of poles of \(f\) is exactly \(\{(x, y, z, w) \mid y = 0 \text{ and } w = 0\}\).
\end{dang}
\textbf{Solution:} First we observe that $XW - YZ$ is a irreducible polynomial, because it's a degree one polynomial in 
$k[X,Y,Z][W]\subset \text{Frac}(k[X,Y,Z])[W]$ and the Gauss lemma. In particular 
$I(V)=(XW-YZ)$ is a prime ideal.
Remember the isomorphism 
\begin{equation}
    k[V]_{(0)}\cong\text{Frac}(k[V]) \rightarrow \callO_{V,V}=k(V): 
    \frac{\overline{F}}{\overline{G}}\mapsto \langle D_V(G), \frac{F}{G} \rangle\quad (*).
\end{equation}
Here, we write $\langle U, F/G\rangle$ for the equivalence class of $(U, F/G)$ in $\callO_{X,X}$.
so, by definition, we have $f=\langle D_V(Y), X/Y\rangle = \langle D_V(W), Z/W\rangle$. Now, write 
$f=\{(U_i, f_i):i\in I\}$, then by definition the domain of $f$ is the set $U=\bigcup_{i\in I}U_i$ and the set of poles 
of $f$ is $P=V\setminus U$. First we show that $P=\{(x,0,z,0)\in V :x,z\in k\}$. Suppose that 
$f=\langle D_V(G), F/G\rangle$ (By isomorphisms $(*)$, any representative has this expression), then we have
\begin{equation}
    YF - XG \in I(V)=(XW - YZ)\;(**).
\end{equation}
Equation $(**)$ implies that $XG \in (Y,W)$, which is a prime ideal ($k[X,Y,Z,W]/(Y,W)$ is isomorphic to 
$k[X,Z]$, which is a domain). Then we 
see that $G\in (Y,W)$, because $X\notin (Y,W)$. In particular $D_V(G)\subset D_V(Y)\cup D_V(W)$ and 
this proves that $U=D_V(Y)\cup D_V(W)$, so 
$P=\{(x,0,z,0)\in k^4\}$ as we claimed. Furthermore, if there exists some representative $(D_V(G), F/G)$
of $f$ such that $U=D_V(G)$, then $G|_{D_V(Y)}\in \callO_V(D_V(Y))^*$, because $D_V(Y)\subset D_V(G)$.\par 
Now we claim that $\callO_V(D_V(Y))^*=k^*\langle Y^m,1/Y^m:m\geq0\rangle$. In fact we have an isomorphism
\begin{equation}
    k[V]_{\overline{Y}}\overset{\cong}{\rightarrow} \callO_V(D_V(Y)): \overline{F}/\overline{Y}^m\mapsto 
    F/Y^m\big|_{D_V(Y)}.
\end{equation}
Furthermore, we have another isomorphism
\begin{equation}
    k[X,Y,W]_Y\rightarrow k[V]_{\overline{Y}}: F/Y^m\mapsto \overline{F}/\overline{Y}^m.
\end{equation}
It's injective because $I(V)\cap k[X,Y,W]=(0)$ and it's surjective because we can use the relation 
$\overline{ZY}=\overline{XW}$ to write any element in $k[V]_{\overline{Y}}$ as a quotient 
$\overline{F}/\overline{Y}^m,\;F\in k[X,Y,W]$. However, $k[X,Y,W]_Y^*=k^*\langle Y^m:m\in\bbZ\rangle$ and we have proved the claim. 
Now, we see that there exist $m\geq1, a\in k^*$ such that $G=aY^m$ on $D_V(Y)$, which means that 
$Y(G-aY^m)|_V=0\iff Y(G-aY^m)\in I(V)\iff G-aY^m\in I(V)$ because $Y\notin I(V)$. Therefore $G=aY^m$ on $V$ and 
$D_V(G)=D_V(Y)$ which is a contradiction, because $(0,0,0,1)\in D_V(W)\setminus D_V(Y)$.\qed

\begin{dang}
\textbf{Exercise 3.} 
Let $X$ and $Y$ be varieties, $P\in X$, and $Q\in Y$. Suppose that the local rings
$\mathcal{O}_{X,P}$ and $\mathcal{O}_{Y,Q}$ are isomorphic (as $k$-algebras).
Show that there exist open neighborhoods $U$ of $P$ in $X$ and $V$ of $Q$ in $Y$
such that $U$ is isomorphic to $V$. Thus, the local ring of a variety at a point
determines the variety, up to isomorphism, in a neighborhood of that point.
\end{dang}
\textbf{Solution:} We can assume without lost of generality that $X\subset k^m$ and $Y\subset k^n$ are affine and irreducible.
Write the isomorphism $\varphi:\callO_{X,P}\to \callO_{Y,Q}$ and write 
$\varphi(\langle X, x_i\rangle)=\langle V_i, g_i\rangle$, where $x_i=X_i|_X$ is the restriction of 
the $i$th coordinate function of $k^m$ to $X$. Since $\varphi$ is an isomorphism, we have 
$\varphi(\fr{m}_{X,P})=\fr{m}_{Y,Q}$, so $\varphi$ descends to an isomorphism
$\overline{\varphi}:\callO_{X,P}/\fr{m}_{X,P}\to \callO_{Y,Q}/\fr{m}_{Y,Q}$. 
So we have the diagram 
\begin{center}
    \begin{tikzcd}
      \callO_{X,P}/\fr{m}_{X,P}\arrow[r,"\overline{\varphi}"]\arrow[d,swap,"\text{ev}_P"]  &\callO_{Y,Q}/\fr{m}_{Y,Q}\arrow[d,"\text{ev}_Q"]\\
      k\arrow[r,dashed] &   k
    \end{tikzcd}
\end{center} 
where the dashed line is a $k$-algebra isomorphism of fields, which must be the identity map. Then we see that 
$\text{ev}_Q \circ\overline{\varphi} = \text{ev}_P$. Now define the morphism 
\begin{equation}
    g: V = \bigcap_i V_i \to k^m: y\mapsto (g_1(y),\dots, g_n(y)).
\end{equation}
We claim that $g(V_0)\subset X$ for some neighborhood $V_0$ of $Q$. In fact, if $F\in I(X)$ we have 
\begin{equation}
    \langle V,F(g_1,\dots,g_m)\rangle = \varphi(\langle X, F(x_1,\dots,x_m))\rangle.
\end{equation}
But $F(x_1,\dots,x_m)=F(X_1,\dots,X_m)|_X=0$, so 
$F(g_1,\dots, g_m)=0$ on some neighborhood $V_0\ni Q, V_0\subset V$. Therefore 
$F(g_1(y),\dots, g_m(y))=0$ and $g(y)\in VI(X)=X$ for $y\in V_0$. Furthermore, we have 
$g_i(Q) = \text{ev}_Q(g_i) = \text{ev}_Q\circ\overline{\varphi}(x_i)=\text{ev}_P(x_i)=x_i(P)$, so 
$g(Q)=P$. For simplicity of notation we will keep writing $V_0=V$. 
Then we see that we constructed a morphism $g:V\to X$ which maps $Q\mapsto P$. In a similar way 
we construct a morphism $f: U\to Y$ which maps $P\mapsto Q$. Now we compose 
\begin{equation}
    V\cap g^{-1}(U) \overset{g}{\rightarrow} U \overset{f}{\rightarrow} X\;(*).
\end{equation}
We claim that this composition is the inclusion. We have 
\begin{equation}
    \varphi(\langle X,x_i\rangle) = \langle V, g_i\rangle,\; 
    \varphi^{-1}(\langle Y, y_i\rangle) = \langle U, f_i\rangle.
\end{equation}
Let $f_i = a_i/b_i$, where $a_i=A_i|_X,b_i = B_i|_X$ for $A_i,B_i\in k[\underline{X}]$. Then we get 
\begin{equation}
    \begin{split}
         \varphi\varphi^{-1}(\langle Y, y_i\rangle) &= \varphi(\langle U, f_i\rangle)\\
         &=\varphi(\langle U,\frac{ A_i(x_1,\dots, x_m)}{B_i(x_1,\dots, x_m)})\rangle)\\
         &=\varphi(\frac{ A_i(\langle U,x_1\rangle,\dots, \langle U,x_m\rangle)}
         {B_i(\langle U,x_1\rangle,\dots,\langle U, x_m\rangle)} )\\
         &=\frac{ A_i(\langle V,g_1\rangle,\dots, \langle V,g_m\rangle)}
         {B_i(\langle V,g_1\rangle,\dots,\langle V, g_m\rangle)} \\
         &=\langle V,\frac{ A_i(g_1,\dots, g_m)}{B_i(g_1,\dots, g_m)}\rangle.
    \end{split}
\end{equation}
So $\frac{ A_i(g_1,\dots, g_m)}{B_i(g_1,\dots, g_m)} = y_i$ on some neighborhood of $V^\prime\ni Q$ in $V$. This proves that 
the composition $(*)$ is the inclusion $V^\prime\hookrightarrow X$. This prove that there exists open sets 
$V\subset Y,U\subset X$ such that $f\circ g = \mathds{1}_V, g\circ f=\mathds{1}_U$, so $g:V\to U$ is an isomorphism.\qed
